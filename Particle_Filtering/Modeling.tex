\subsection{Modeling}
\textbf{System} \textit{Non-linear, discrete-time}:
\begin{align*}
    x(k) &= q_{k-1}(x(k-1), v(k-1)) \\
    z(k) &= h_k(x(k), w(k))
\end{align*}
$x(0), \{v\}, \{w\}$ mutually indep. can be D- or CRVs, known PDFs
\smallsectionbox{Objective} 
Bayesian State Estimator with Monte Carlo Sampling
\begin{align*}
    \begin{array}{l r l}
        \textbf{Init:} &x_m(0) &:= x(0) \\
        \textbf{S1:} &x_p(k) &:= q_{k-1}(x_m(k-1), v(k-1)) \\
        \textbf{S2:} &z_m(k) &:= h_k(x_p(k), w(k)) \\
        &p_{x_m(k)}(\xi) &:= p_{x_p(k)|z_m(k)}(\xi | \bar{z}(k)) \quad \forall\xi
    \end{array}
\end{align*}
Similar proof for $p_{x_p(k)}(\xi), p_{x_m(k)}(\xi) \forall\xi$ as for KF aux. var.
