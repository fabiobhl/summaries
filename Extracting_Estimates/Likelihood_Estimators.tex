\subsection{General Measurement}
\begin{align*}
    \underbrace{
    \begin{bmatrix}
        z_1 \\
        z_m
    \end{bmatrix}
    }_{z \ \in \ \mathbb{R}^m}
    = 
    \underbrace{
        \begin{bmatrix}
            h_{11} & h_{1n} \\
            h_{m1} & h_{mn}
        \end{bmatrix}
    }_{H \ \in \ \mathbb{R}^{m\times n}}
    \cdot
    \underbrace{
        \begin{bmatrix}
            x_1 \\
            x_n
        \end{bmatrix}
    }_{x \ \in \ \mathbb{R}^n}
    +
    \underbrace{
        \begin{bmatrix}
            w_1 \\
            w_m
        \end{bmatrix}
    }_{w \ \in \ \mathbb{R}^m}
\end{align*}
$m$ Measurements, $n$ States, $w_i \sim \mathcal{N}(0,R)$ -- Zero-mean additive gaussian, $x,w_i \ \forall i$ assumed mutually independent
\subsection{Likelihood Estimators}
\subsubsection{ML-Maximum Likelihood}
Choose $x$ to make $\bar{z}$ most likely:
\begin{align*}
    \begin{drcases}
        x\in\mathcal{X} & \mathrm{unknown} \\
        p_x(\bar{x}) & \mathrm{unknown}
    \end{drcases}
    \quad
    \widehat{x}^{ML} := \mathop{\mathrm{argmax}}_{\bar{x}\in\mathcal{X}}
    \left[ p_{z|x}(\bar{z}|\bar{x}) \right]
\end{align*}
$p_{z|x}(\bar{z}|\bar{x})$ is var. --  technically not a PDF
\subsubsection{MAP-Maximum A Posteriori}
ML given observation \& prior belief about $x$:
\begin{align*}
    \begin{drcases}
        x\in\mathcal{X} & \mathrm{unknown} \\
        p_x(\bar{x}) & \textbf{known}
    \end{drcases}
    \quad
    \widehat{x}^{MAP} := \mathop{\mathrm{argmax}}_{\bar{x}\in\mathcal{X}}
    \left[ p_{z|x}(\bar{z}|\bar{x}) p_x(\bar{x}) \right]
\end{align*}
$\hat{x}^{MAP} = \hat{x}^{ML}$ for constant $p_x(\bar{x})$
