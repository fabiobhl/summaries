\subsection{Kalman Filter}
\smallsectionbox{Model (Linear Time-Invariant)}

\begin{align*}
    x(k) &= A(k-1)x(k-1)+u(k-1)+v(k-1)\\
    z(k) &= H(k)x(k)+w(k)
\end{align*}
\vspace{-1.8em}
\begin{center}
    \begin{tabular}{|l|l|}
        \hline
        $x(0), v(\cdot), w(\cdot)$ ind. & $x(0) \sim \mathcal{N}(x_0, P_0)$\\
        \hline
        $v \sim \mathcal{N}(0, Q(k))$ & $w \sim \mathcal{N}(0, R(k))$\\
        \hline
    \end{tabular} 
\end{center}
\vspace{-0.8em}

\smallsectionbox{Auxilliary Variables}
\textbf{Problem Formulation} want $p_{x(k)|z(1:k)} \leadsto$ bayesian tracking provides this
\textbf{BUT} we have \textbf{CRVs} and don't want integrals $\Rightarrow$ exploit linearity\\
\textbf{Auxilliary Variables:}
\begin{align*}
        \textbf{Init: } x_m(0) &= x(0) \\
        \textbf{S1: } x_p(k) &= A(k-1)x_m(k-1) + u(k-1) + v(k-1) \\
        \textbf{S2: } z_m(k) &= H(k) x_p(k) + w(k) \\
        p_{x_m(k)}(\xi) &:= p_{x_p(k)|z_m(k)}(\xi | \bar{z}(k)) \quad \forall\xi
\end{align*}
\textbf{Equivalency}:
\begin{align*}
    p_{x_p(k)}(\xi) &= p_{x(k)|z(1:k-1)}(\xi|\bar{z}(1:k-1)) \\
    p_{x_m(k)}(\xi) &= p_{x(k)|z(1:k)}(\xi|\bar{z}(1:k))
\end{align*}
\textbf{GRV} $x_p$ and $x_m$ are GRVs for $k=1,2,\dots$\\
\begin{minipage}[c]{0.45\linewidth}
    \begin{align*}
        \widehat{x}_p(k) &:= \mathbb{E}\left[ x_p(k) \right] \\
        \widehat{x}_m(k) &:= \mathbb{E}\left[ x_m(k) \right]
    \end{align*}
\end{minipage}
\begin{minipage}[c]{0.45\linewidth}
    \begin{align*}
        P_p(k) &:= \mathrm{Var}\left[ x_p(k) \right] \\
        P_m(k) &:= \mathrm{Var}\left[ x_m(k) \right]
    \end{align*}
\end{minipage}

\smallsectionbox{Recursive Update Equations}
\textbf{Initialization} $\quad\widehat{x}_m(0) = x_0, \quad P_m(0) = P_0$ \\
\textbf{S1 -- Prior Update / Prediction Step}
\begin{align*}
    \widehat{x}_p(k) &= A(k-1) \widehat{x}_m(k-1) + u(k-1) \\
    P_p(k) &= A(k-1)P_m(k-1)A^\top(k-1) + Q(k-1)
\end{align*}
\textbf{S2 -- A Posteriori Update / Measurement Update}
\begin{align*}
    P_m(k) &= \left( P_p^{-1}(k) + H^\top(k) R^{-1}(k) H(k) \right)^{-1}\\
    \widehat{x}_m(k) &= \widehat{x}_p(k) + 
    \underbrace{P_m(k) H^\top(k) R^{-1}(k)}_{\mathrm{correction factor}}
    \underbrace{\left( \bar{z}(k) - H(k) \widehat{x}_p(k) \right)}_{\mathrm{prediction error}}
\end{align*}

\smallsectionbox{Alternative Equations -- all matrices/vectors at time k}
\begin{equation*}
    K(k) = P_p(k) H^\top(k) \left( H(k) P_p(k) H^\top(k) \!+\! R(k) \right)^{-1} 
\end{equation*}
\textbf{S2 -- Kalman Filter Gain:}
\begin{align*}
    \widehat{x}_m &= \widehat{x}_p + K \left( \bar{z} \!-\! H \widehat{x}_p \right) \\
    P_m &= \underbrace{\left( \mathbb{I} - K H \right) P_p}_{\textit{cheaper}}
    = \underbrace{\left( \mathbb{I} - K H \right) P_p \left( \mathbb{I} - K H \right)^\top + K R K^\top}_{\textit{less sensitive to num. errors}}
\end{align*}

\smallsectionbox{KF Remarks}
\begin{itemize}[leftmargin=5pt]
    \item The KF is the analytical solution to the Bayesian state estimation problem for linear systems with Gaussian distributions. (Best You can do)
    \item Otherwise, the KF is no longer optimal. It is the best linear estimator in the MMSE sense but nonlinear might do better.
    \item $A(k),H(k),Q(k),R(k) \ \forall k$ as well as $P_0$ $\leadsto$ KF matrices $P_p(k), P_m(k), K(k)$ can be computed offline, $K$ no f() of Measurm.
    \item KF is linear and time-varying:
    \begin{align*}
        &\widehat{x}_m(k) = \tilde{A}(k) \cdot \widehat{x}_m(k-1) + \tilde{B}(k) \cdot u(k-1) + K\bar{z}(k) \\
        &\tilde{A}(k) = (\mathbb{I} - K(k)H(k)) A(k-1), \quad \tilde{B}(k) = (\mathbb{I} - K(k)H(k))
    \end{align*}
    \item Among class of linear, unbiased estimators, the KF minimizes MSE
    \item Not all system matrices need to be pd. some can also be psd. as long as inversions are possible in the update equations.
\end{itemize}

\subsubsection{Detectability / Stabilizablilty}
\smallsectionbox{Detectability Conditions}
Pair $(A,H)$ detectable
\begin{itemize}[label=$\Leftrightarrow$, leftmargin=1.6em]
    \item For deterministic LTI Sys. ($\scriptstyle x(k) = Ax(k-1), z(k) = Hx(k)$) \\
    $\lim_{k\to\infty}z(k)=0 \Rightarrow \lim_{k\to\infty}x(k)=0 \
    \quad \forall x_0 \in \mathbb{R}^n$
    \item $\left[\begin{smallmatrix}
        A - \lambda \mathbb{I} \\
        H
    \end{smallmatrix}\right]$ is full rank for all $\lambda \in \mathbb{C}$ 
    with $\lvert \lambda\rvert \geq 1$ (PBH-Test)
    \item Eigenvalues of $A-LH$ (or $(\mathbb{I}-LH)A$) can be placed within unit circle
    by suitable choice of $L\in\mathbb{R}^{n\times m}$
\end{itemize}

\smallsectionbox{Stabilizability Conditions}
Pair $(A,B)$ stabilizable
\begin{itemize}[label=$\Leftrightarrow$, leftmargin=1.6em]
    \item For deterministic LTI Sys. ($\scriptstyle x(k) = Ax(k-1) + Bu(k-1)$) \\
    $\exists \ u(0:k-1) \mathrm{ s.t. } \lim_{k\to\infty}x(k)=0 
    \quad \forall x_0 \in \mathbb{R}^n$
    \item $\left[\begin{smallmatrix}
        A - \lambda \mathbb{I} & B
    \end{smallmatrix}\right]$ full rank $\forall \lambda \in \mathbb{C}$ 
    with $\lvert \lambda\rvert \geq 1$ (PBH-Test)
    \item Eigenvalues of $A-BK$ (or $(\mathbb{I}-BK)A$) can be placed within unit circle
    by suitable choice of $K\in\mathbb{R}^{m\times n}$
\end{itemize}

\smallsectionbox{Observability / Reachability}
if $(A^\top, B^\top)$ detectable\\
\begin{minipage}[c]{0.40\linewidth}
    \textbf{Observability Matrix} \\
    implies detectability
\end{minipage}
\begin{minipage}[c]{0.3\linewidth}
    $\mathcal{O} =  \left[\begin{smallmatrix}
        H \\
        HA^{n-1}
    \end{smallmatrix}\right]$
\end{minipage}

\begin{minipage}[c]{0.40\linewidth}
    \textbf{Reachability Matrix} \\
    implies stabilizablilty
\end{minipage}
\begin{minipage}[c]{0.3\linewidth}
    $\mathcal{R} = \left[\begin{smallmatrix}
        A &
        A^{n-1}B
    \end{smallmatrix}\right]$
\end{minipage}

\subsubsection{Asymptotic Properties of KF}
\textbf{Model} assume \textbf{TI System, stationary Distributions} ($A, H, Q, R$ constants) \\
$\leadsto$ the KF is still time-varying!\\
\textbf{Change of Variance}
\begin{align*}
    P_p({\scriptstyle k+1}) &= \left(A  P_p({\scriptstyle k}) A^\top\right) + Q \\
    &- \left( A P_p({\scriptstyle k}) H^\top \right) 
    \left( H  P_p({\scriptstyle k}) H^\top \!\!+\! R \right)^{-1} 
    \left( H P_p({\scriptstyle k}) A^\top\right)
\end{align*}
\textbf{Scalar Case} $P_p(k+1) = \frac{a^2 r P_p(k)}{h^2 P_p(k) + r}+q =: f(P_p(k))$ \\
\textbf{Summary} 
\textit{Variance Converges to unique steady-state sol'n \\
provided either $\lvert a\rvert <1$ or if $\lvert a \rvert \geq 1$ $\Rightarrow$ 
$h \neq 0$ and $q >0$}\\

\subsubsection{Steady-State KF}
\textbf{DARE -- Discrete Algebraic Ricatti Equation} ($P_\infty$ is $P_p$)
\begin{equation*}
    P_\infty = A P_\infty A^\top + Q 
    - AP_\infty H^\top 
    (H P_\infty H^\top + R)^{-1}
    H P_\infty A^\top
\end{equation*}

\textbf{Steady-State KF Gain: } $K_\infty = P_\infty H^\top (H P_\infty H^\top \!\!+\! R)^{-1}$


\smallsectionbox{Steady State KF Equations $\widehat{x}(k) := \widehat{x}_m(k)$}

\begin{align*}
    \widehat{x}(k) = 
    (\mathbb{I} \!-\! K_\infty H) A \widehat{x}({\scriptstyle k - 1}) + 
    (\mathbb{I} \!-\! K_\infty H) u({\scriptstyle k - 1}) \!+\! K_\infty \bar{z}(k)\\
    \widehat{x}(k) =  \widehat{A} \cdot \widehat{x}({\scriptstyle k - 1}) + \widehat{B} \cdot u({\scriptstyle k - 1}) + K_\infty \cdot \bar{z}(k), \quad \widehat{x}(0) = x_0
\end{align*}

\smallsectionbox{Error dynamics (also holds for non ss-KF)  $e(k) = x(k) - x_f(k)$}

\begin{align*}
    &e(k) = (\mathbb{I} \!-\! K_\infty H)A e(k\!-\!\!1) 
    \!+\! (\mathbb{I} \!-\! K_\infty H) v(k\!-\!1) \!-\! K_\infty w({\scriptstyle k}) \\
    &\mathbb{E}[e(k)] \!=\! (\mathbb{I} \!-\! K_\infty H)A \ \mathbb{E}[e(k\!-\!1)]
\end{align*}
\begin{itemize}[leftmargin=1em, itemsep=0pt]
    \item If $\mathbb{E}[e(0)] = 0$ then $\mathbb{E}[e(k)] = 0 \quad \forall k$ $\leadsto$ KF is unbiased
    \item $\mathbb{E}[e(k)] \xrightarrow{k\to\infty} 0$ $\Leftrightarrow$ $(\mathbb{I}-K_\infty H)A$ stable (all $\lvert\lambda\rvert < 1$)
\end{itemize}

\textbf{Theorem} for $R>0, Q \geq 0, G \text{ s.t. } Q=GG^\top$
    \begin{itemize}[label=$\Leftrightarrow$, leftmargin=1.6em]
        \item $(A,H)$ is detectable and $(A,G)$ is stabilizable
        \item DARE has unique pos. semidef. sol. $P_\infty \geq 0$, 
        resulting $(\mathbb{I} - K_\infty H)A$ is stable and
        \begin{align*}
            \lim_{k\to\infty} P_p(k) = P_\infty \quad \forall P_p(1) \geq 0 \text{ and } \forall P_m(0) = P_0 \geq 0
        \end{align*}
    \end{itemize}
\begin{description}
    \item[(A,H) Detectable:] can observe all unstable modes 
    \item[(A,G) Stabilizable:] noise excites unstable modes
\end{description}
\textbf{Note} if $Q>0 \Rightarrow (A,G)$ always stabilizable
If $Q=0$ multiple valid solutions can be possible
