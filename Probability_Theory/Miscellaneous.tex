    \textbf{Indep. multiplication:} $z = xy \to p(z) = \sum_{Y} p_x(\frac{z}{y})p_y(y)$ \\
    \smallsectionbox{Uniform Distribution}
    \begin{align*}
        p(x) = \begin{cases}
            \frac{1}{b-a} & a \leq x \leq b \\
            0 & \mathrm{else}
        \end{cases}
        \qquad\left\lvert\qquad
        \begin{aligned}
            \mathbb{E}\{x\} &= \textstyle\frac{a+b}{2} \\
            \mathrm{Var}\{x\} &= \textstyle\frac{(b-a)^2}{12}
        \end{aligned}
        \right.
    \end{align*}
    \smallsectionbox{Triangular Distribution}
    \begin{align*}
        p(x) = \begin{cases}
            \frac{2(x-a)}{(b-a)(c-a)} & a \leq x \leq c \\
            \frac{2(b-x)}{(b-a)(b-c)} & c \leq x \leq b \\
            0 & \mathrm{else}
        \end{cases}
    \end{align*}
    \begin{align*}
        F(x) = \begin{cases}
            0 & x < a \\
            \frac{(x-a)^2}{(b-a)(c-a)} & a \leq x \leq c \\
            1-\frac{(b-x)^2}{(b-a)(b-c)} & c \leq x \leq b \\
            1 & x > b
        \end{cases}
    \end{align*}
    \begin{align*}
            \mathbb{E}\{x\} = \textstyle\frac{a+b+c}{3}, \quad
            \mathrm{Var}\{x\} = \textstyle\frac{a^2+b^2+c^2-ab-ac-bc}{18}
    \end{align*}

    \smallsectionbox{Gaussian Distribution $\mathbf{x} = (x_1, \dots, x_D) \sim \mathcal{N}(\boldsymbol{\mu}, \mathbf{\Sigma})$}
    \begin{align*}
        p_{\mathbf{x}}(\mathbf{\bar{x}}) = \frac{1}{(2\pi)^{D/2} \det(\mathbf{\Sigma})^{1/2}}\exp\left(-\frac{(\mathbf{\bar{x}}-\boldsymbol{\mu})^T\mathbf{\Sigma}^{-1}(\mathbf{\bar{x}}-\boldsymbol{\mu})}{2}\right)
    \end{align*}
    $\leadsto$ fully characterized by mean $\boldsymbol{\mu} \in \mathbb{R}^D$ and Variance $\boldsymbol{\Sigma} \in \mathbb{R}^{D\times D}$\\
    (Scalar: $D = 1, \ \sigma^2 = \det(\Sigma)\to \det(\Sigma)^{1/2} = \sigma$)
    \begin{align*}
        p(x) = \frac{1}{\sqrt{2\pi\sigma^2}}\exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right)
        \qquad\left\lvert\qquad
        \begin{aligned}
            \mathbb{E}\{x\} &= \mu \\
            \mathrm{Var}\{x\} &= \sigma^2
        \end{aligned}
        \right.
    \end{align*}
    \textbf{Affine transformation of Gaussian: $\boldsymbol{x} \sim \mathcal{N}(\boldsymbol{\mu},\boldsymbol{\Sigma})$}
    \begin{align*}
        \boldsymbol{y} = M \boldsymbol{x} + \boldsymbol{b} \quad \to \quad \boldsymbol{y} \sim \mathcal{N}(M \boldsymbol{\mu} + \boldsymbol{b}, M \boldsymbol{\Sigma} M^T)
    \end{align*}
    \textbf{Linear combination of jointly GRV: $\boldsymbol{x}, \boldsymbol{y}$}
    \begin{align*}
        &\boldsymbol{z} = M_x \boldsymbol{x} + M_y \boldsymbol{y} \quad \Rightarrow \quad \boldsymbol{z} \sim \mathcal{N}(\boldsymbol{\mu}_z, \boldsymbol{\Sigma}_z)\\
        &\boldsymbol{\mu}_z = M_x \boldsymbol{\mu}_x + M_y \boldsymbol{\mu}_y \\
        &\boldsymbol{\Sigma}_z = M_x \boldsymbol{\Sigma}_x M_x^T + M_x \textcolor{red}{\boldsymbol{\Sigma}_{xy}} M_y + M_y \textcolor{red}{\boldsymbol{\Sigma}_{xy}^T} M_x + M_y \boldsymbol{\Sigma}_y M_y^T
    \end{align*}
    (\textcolor{red}{Red symbols are zero} if $\boldsymbol{x}, \boldsymbol{y}$ are independent)\\
    \textbf{Diagonal Variance} $\leadsto \ y_i$ mut.indep. iff $\Sigma$ diagonal
    \begin{equation*}
        p(y) = \prod_{i=1}^{D} \textstyle\frac{1}{\sqrt{2\pi\sigma_i^2}} 
        \exp(-\frac{(y_i - \mu_i)^2}{2\sigma_i^2})
    \end{equation*}
