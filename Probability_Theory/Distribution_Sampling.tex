\subsection{Distribution Sampling}
    \textbf{CDF -- Cumulative Distribution Function} 
    \begin{equation*}
        \widehat{F}_x(\bar{x}) := \textstyle\sum_{i=-\infty}^{\bar{x}}\widehat{p}_x(i) = Pr(x \leq \bar{x}) = \int_{-\infty}^{\hat{x}} p_x(\hat{x}) d\hat{x}
    \end{equation*}

    \textbf{A1 -- 1 DRV} $\bar{u}$ sample of $u\sim\mathcal{U}$, solve for $\bar{x}$ such that:
    \begin{equation*}
        \widehat{F}_x(\bar{x}-1) < \bar{u} \leq \widehat{F}_x(\bar{x})  
    \end{equation*}
